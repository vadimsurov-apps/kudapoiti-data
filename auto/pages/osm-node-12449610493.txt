document.addEventListener("DOMContentLoaded", function () {
// Получаем все вкладки из обоих блоков
const tabs = document.querySelectorAll(".t-tabs__item");
const contents = document.querySelectorAll(".t-tabs__content");
// Функция переключения вкладок
function switchTab(tabIndex) {
// Удаляем активные состояния у всех вкладок и скрываем содержимое
tabs.forEach(tab => tab.classList.remove("t-tabs__item_active"));
contents.forEach(content => content.style.display = "none");
// Активируем нужную вкладку и показываем соответствующий контент
tabs.forEach((tab, index) => {
if (index === tabIndex) {
tab.classList.add("t-tabs__item_active");
});
if (contents[tabIndex]) {
contents[tabIndex].style.display = "block";
// Добавляем обработчики клика для всех вкладок
tabs.forEach((tab, index) => {
tab.addEventListener("click", (event) => {
event.preventDefault(); // Отключаем переход по ссылке (если вкладка оформлена ссылкой)
switchTab(index);
});
// Добавляем touch-событие для мобильных устройств
tab.addEventListener("touchstart", (event) => {
event.preventDefault(); // Отключаем стандартное поведение
switchTab(index);
});
});
});
